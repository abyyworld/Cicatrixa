The proof: the test it wrote

Written by the reproducer and run against the code as it was. It failed — the bug was proven — at 12:11:25 PM, before the fixer changed a line.

tests/test_create_order_full_discount.py
import main
from main import OrderRequest


def reset_order_state() -> None:
    """Reset the mutable globals so tests remain deterministic."""
    main.orders.clear()
    main.order_counter = 0


def test_create_order_all_discount_yields_zero_total() -> None:
    reset_order_state()
    request = OrderRequest(item="demo", quantity=1, price_cents=1000, discount_pct=100)

    response = main.create_order(request)

    # After a 100% discount the total charge should be zero cents.
    assert response.total_cents == 0, "Full discounts should return zero total, not crash."
    assert response.discount_applied == 100
    assert main.orders[response.order_id]["total_cents"] == 0
